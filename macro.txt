Sub ConvertShapesToPlaceInCell()
    Dim shp As Shape
    Dim targetCell As Range
    Dim ws As Worksheet
    
    Set ws = ActiveSheet
    
    ' Percorre todas as formas/imagens na aba
    For Each shp In ws.Shapes
        ' Verifica se é uma imagem (Picture)
        If shp.Type = msoPicture Then
            ' Pega a célula que está no canto superior esquerdo da imagem
            Set targetCell = shp.TopLeftCell
            
            ' Seleciona a imagem
            shp.Select
            
            ' Executa o comando de interface que move a imagem para dentro da célula
            ' Nota: Este comando simula o clique com o botão direito > Place in Cell
            On Error Resume Next
            Application.CommandBars.ExecuteMso "PicturePlaceInCell"
            On Error GoTo 0
        End If
    Next shp
    
    MsgBox "Conversão concluída!", vbInformation
End Sub
